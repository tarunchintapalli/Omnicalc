function cssVars(t) {
  return {
    "--bg": t.bg, "--surface": t.surface, "--surface-2": t.surface2,

const inputClass =
  "w-full rounded-lg border border-[var(--border)] bg-[var(--surface-2)] px-3.5 py-2.5 text-[15px] text-[var(--ink)] outline-none transition focus:border-[var(--brass)] focus:ring-2 focus:ring-[var(--brass-soft)]";

function Field({ label, hint, children }) {
  return (
    <label className="block">
      <span className="mb-1.5 flex items-baseline justify-between">
        <span className="text-[11px] font-semibold uppercase tracking-[0.12em] text-[var(--ink-muted)]">{label}</span>
        {hint && <span className="text-[11px] text-[var(--ink-muted)]">{hint}</span>}
      </span>
      {children}
    </label>
  );
}

function NumberInput({ value, onChange, suffix, ...rest }) {
  return (
    <div className="relative">
      <input
        type="number"
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className={inputClass + (suffix ? " pr-10" : "")}
        {...rest}
      />
      {suffix && (
        <span className="pointer-events-none absolute right-3.5 top-1/2 -translate-y-1/2 text-sm text-[var(--ink-muted)]">
          {suffix}
        </span>
      )}
    </div>
  );
}

function SelectInput({ value, onChange, options }) {
  return (
    <select value={value} onChange={(e) => onChange(e.target.value)} className={inputClass}>
      {options.map((o) => (
        <option key={o.value} value={o.value}>{o.label}</option>
      ))}
    </select>
  );
}

function Segmented({ options, value, onChange, full }) {
  return (
    <div className={`inline-flex rounded-lg border border-[var(--border)] bg-[var(--surface-2)] p-1 ${full ? "w-full" : ""}`}>
      {options.map((opt) => (
        <button
          key={opt.value}
          type="button"
          onClick={() => onChange(opt.value)}
          className={`${full ? "flex-1" : ""} rounded-md px-3.5 py-1.5 text-sm transition ${
            value === opt.value
              ? "bg-[var(--brass)] font-semibold text-[var(--surface)] shadow-sm"
              : "text-[var(--ink-muted)] hover:text-[var(--ink)]"
          }`}
        >
          {opt.label}
        </button>
      ))}
    </div>
  );
}

function Slider({ value, onChange, min, max, step }) {
  const pct = ((value - min) / (max - min)) * 100;
  return (
    <input
      type="range"
      min={min}
      max={max}
      step={step}
      value={value}
      onChange={(e) => onChange(e.target.value)}
      className="w-full accent-[var(--brass)]"
      style={{
        background: `linear-gradient(90deg, var(--brass) ${pct}%, var(--border) ${pct}%)`,
        height: 6, borderRadius: 999, appearance: "none", outline: "none",
      }}
    />
  );
}

function Card({ children, className = "" }) {
  return (
    <div className={`rounded-2xl border border-[var(--border)] bg-[var(--surface)] p-6 shadow-[0_1px_2px_rgba(0,0,0,0.04)] ${className}`}>

  );
}

function DigitalReadout({ label, value, sub, big }) {
  return (
    <div className="rounded-xl border border-black/20 bg-[var(--display-bg)] px-5 py-4 shadow-inner">
      <div className="mb-1 text-[10px] uppercase tracking-[0.22em]" style={{ color: "var(--display-ink)", opacity: 0.6 }}>
        {label}
      </div>
      <div
        className={`font-mono tracking-tight ${big ? "text-3xl sm:text-4xl" : "text-2xl"}`}

      </div>

}

function SplitBar({ a, b, aLabel, bLabel, aColor = "var(--brass)", bColor = "var(--teal)" }) {
  const total = a + b || 1;
  const pa = Math.max(0, Math.min(100, (a / total) * 100));
  return (
    <div>
      <div className="flex h-3 w-full overflow-hidden rounded-full border border-[var(--border)] bg-[var(--surface-2)]">
        <div style={{ width: `${pa}%`, background: aColor }} />
        <div style={{ width: `${100 - pa}%`, background: bColor }} />
      </div>
      <div className="mt-2 flex justify-between text-xs text-[var(--ink-muted)]">
        <span><span className="mr-1.5 inline-block h-2 w-2 rounded-full align-middle" style={{ background: aColor }} />{aLabel}</span>
        <span><span className="mr-1.5 inline-block h-2 w-2 rounded-full align-middle" style={{ background: bColor }} />{bLabel}</span>
      </div>
    </div>
  );
}

function BmiGauge({ bmi }) {
  const min = 10, max = 45;
  const pct = Math.max(0, Math.min(100, ((bmi - min) / (max - min)) * 100));
  return (
    <div>
      <div className="flex h-3 w-full overflow-hidden rounded-full">
        <div style={{ flex: 8.5, background: "var(--blue)" }} />
        <div style={{ flex: 6.4, background: "var(--green)" }} />
        <div style={{ flex: 5, background: "#D9A544" }} />
        <div style={{ flex: 15, background: "var(--red)" }} />
      </div>
      <div className="relative h-3">
        <div
          className="absolute -top-1 h-3 w-[3px] rounded bg-[var(--ink)] transition-all"
          style={{ left: `calc(${pct}% - 1.5px)` }}
        />
      </div>
      <div className="mt-1 flex justify-between text-[10px] text-[var(--ink-muted)]">
        <span>10</span><span>18.5</span><span>25</span><span>30</span><span>45</span>
      </div>
    </div>
  );
}

/* ======================================================================
   FORMAT HELPERS
   ====================================================================== */
const fmt = (n, digits = 2) => {
  if (!isFinite(n)) return "0";
  return n.toLocaleString("en-US", { minimumFractionDigits: digits, maximumFractionDigits: digits });
};
const money = (n) => "$" + fmt(n, 2);
const toNum = (v, fallback = 0) => {
  const n = parseFloat(v);
  return isNaN(n) ? fallback : n;
};

/* ======================================================================
   FINANCIAL: EMI
   ====================================================================== */
function EmiCalc() {
  const [principal, setPrincipal] = useState("250000");
  const [rate, setRate] = useState("7.5");
  const [years, setYears] = useState("15");

  const { emi, totalInterest, totalPayment } = useMemo(() => {

    const e = r === 0 ? P / n : (P * r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);

          </div>
        </Card>

        <p>Equated Monthly Installment uses the reducing-balance formula, where each payment covers interest on the outstanding balance plus a portion of principal.</p>

        <p>P is the loan principal, r is the monthly interest rate (annual rate ÷ 12 ÷ 100), and n is the total number of monthly installments.</p>
        <p className="text-xs">Example: a $250,000 loan at 7.5% for 15 years produces a fixed monthly payment that stays constant even as the interest/principal mix shifts over time.</p>
      </ExplainSection>

/* ======================================================================
   FINANCIAL: SIP
   ====================================================================== */
function SipCalc() {
  const [monthly, setMonthly] = useState("500");
  const [rate, setRate] = useState("12");
  const [years, setYears] = useState("10");

        <DigitalReadout label="Total Wealth Value" value={money(future)} big />

        <p>A Systematic Investment Plan compounds a fixed monthly contribution at an assumed rate of return, similar to an annuity due.</p>

        <p>M is the monthly investment, i is the monthly rate of return, and n is the total number of installments. Returns are estimates — actual market performance varies.</p>
      </ExplainSection>
    </div>

}

/* ======================================================================
   FINANCIAL: COMPOUND INTEREST

function CompoundCalc() {

  const [years, setYears] = useState("10");

          <div className="grid grid-cols-2 gap-4 text-sm">

      </ExplainSection>
    </div>

/* ======================================================================
   HEALTH: BMI
   ====================================================================== */
function BmiCalc() {
  const [unit, setUnit] = useState("metric");
  const [heightCm, setHeightCm] = useState("175");
  const [weightKg, setWeightKg] = useState("72");
  const [heightFt, setHeightFt] = useState("5");
  const [heightIn, setHeightIn] = useState("9");

      const h = toNum(heightCm) / 100, w = toNum(weightKg);
      return h > 0 ? w / (h * h) : 0;
    } else {
      const totalIn = toNum(heightFt) * 12 + toNum(heightIn), w = toNum(weightLb);
      return totalIn > 0 ? (703 * w) / (totalIn * totalIn) : 0;

  }, [unit, heightCm, weightKg, heightFt, heightIn, weightLb]);

  const category = bmi < 18.5 ? "Underweight" : bmi < 25 ? "Normal" : bmi < 30 ? "Overweight" : "Obese";

        <DigitalReadout label="Imperial" value="BMI = 703 × lb / in²" />
        <p>BMI is a screening tool, not a diagnostic measure — it doesn't account for muscle mass, bone density, or body composition.</p>
      </ExplainSection>
    </div>
  );
}

/* ======================================================================
   HEALTH: IDEAL BODY WEIGHT
   ====================================================================== */
function IbwCalc() {
  const [gender, setGender] = useState("male");
  const [heightCm, setHeightCm] = useState("175");
  const [frame, setFrame] = useState("medium");

  const { devine, miller } = useMemo(() => {
    const totalIn = toNum(heightCm) / 2.54;
    const overIn = Math.max(0, totalIn - 60);
    let d = gender === "male" ? 50 + 2.3 * overIn : 45.5 + 2.3 * overIn;
    let m = gender === "male" ? 56.2 + 1.41 * overIn : 53.1 + 1.36 * overIn;
    const factor = frame === "small" ? 0.9 : frame === "large" ? 1.1 : 1;
    return { devine: d * factor, miller: m * factor };
  }, [gender, heightCm, frame]);

  return (
    <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
      <Card>
        <div className="space-y-5">

          <Field label="Frame Size">
            <Segmented full options={[{ value: "small", label: "Small" }, { value: "medium", label: "Medium" }, { value: "large", label: "Large" }]} value={frame} onChange={setFrame} />
          </Field>
        </div>
      </Card>

      <div className="space-y-4">
        <DigitalReadout label="Devine Formula" value={`${fmt(devine, 1)} kg`} sub={`≈ ${fmt(devine * 2.20462, 1)} lb`} big />
        <Card>
          <div className="text-sm text-[var(--ink-muted)]">Miller Formula</div>
          <div className="text-lg font-semibold">{fmt(miller, 1)} kg <span className="text-sm font-normal text-[var(--ink-muted)]">(≈ {fmt(miller * 2.20462, 1)} lb)</span></div>
        </Card>
      </div>

      <ExplainSection title="How ideal body weight is estimated">
        <p>Both formulas start from a baseline weight at 5 feet tall and add weight per inch above that height, then adjust for body frame size.</p>

        <DigitalReadout label="Miller (male)" value="56.2 kg + 1.41 kg / in over 5ft" />

      </ExplainSection>
    </div>
  );
}

/* ======================================================================
   MATH: PERCENTAGE
   ====================================================================== */
function PercentageCalc() {
  const [mode, setMode] = useState("of");
  const [x, setX] = useState("20");
  const [y, setY] = useState("150");

  const result = useMemo(() => {
    const X = toNum(x), Y = toNum(y);
    if (mode === "of") return { value: (X / 100) * Y, label: `${X || 0}% of ${Y || 0}` };
    if (mode === "isWhat") return { value: Y !== 0 ? (X / Y) * 100 : 0, label: `${X || 0} is what % of ${Y || 0}` };
    const change = X !== 0 ? ((Y - X) / X) * 100 : 0;
    return { value: change, label: `Change from ${X || 0} to ${Y || 0}` };
  }, [mode, x, y]);

  return (
    <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
      <Card>
        <div className="space-y-5">
          <Field label="Mode">
            <Segmented
              full
              options={[
                { value: "of", label: "X% of Y" },
                { value: "isWhat", label: "X is % of Y" },
                { value: "change", label: "% change" },
              ]}
              value={mode}
              onChange={setMode}
            />
          </Field>
          <Field label={mode === "of" ? "X (percent)" : mode === "isWhat" ? "X (value)" : "X (from)"}>
            <NumberInput value={x} onChange={setX} suffix={mode === "of" ? "%" : ""} />
          </Field>
          <Field label={mode === "of" ? "Y (total)" : mode === "isWhat" ? "Y (total)" : "Y (to)"}>
            <NumberInput value={y} onChange={setY} />
          </Field>
        </div>
      </Card>

      <div className="space-y-4">
        <DigitalReadout
          label={result.label}
          value={mode === "isWhat" || mode === "change" ? `${fmt(result.value, 2)}%` : fmt(result.value, 2)}
          big
        />
        <Card>
          <p className="text-sm text-[var(--ink-muted)]">
            {mode === "of" && `${x || 0}% of ${y || 0} equals ${fmt(result.value, 2)}.`}
            {mode === "isWhat" && `${x || 0} is ${fmt(result.value, 2)}% of ${y || 0}.`}
            {mode === "change" && `Moving from ${x || 0} to ${y || 0} is a ${result.value >= 0 ? "increase" : "decrease"} of ${fmt(Math.abs(result.value), 2)}%.`}
          </p>
        </Card>
      </div>

      <ExplainSection title="Percentage formulas">
        <DigitalReadout label="X% of Y" value="(X ÷ 100) × Y" />
        <DigitalReadout label="X is what % of Y" value="(X ÷ Y) × 100" />
        <DigitalReadout label="% change X → Y" value="((Y − X) ÷ X) × 100" />
      </ExplainSection>
    </div>
  );
}

/* ======================================================================
   MATH: SCIENTIFIC CALCULATOR
   ====================================================================== */
function ScientificCalc() {
  const [display, setDisplay] = useState("0");
  const [stored, setStored] = useState(null);
  const [operator, setOperator] = useState(null);
  const [waiting, setWaiting] = useState(false);
  const [memory, setMemory] = useState(0);

  const inputDigit = (d) => {
    if (waiting) { setDisplay(d); setWaiting(false); return; }
    setDisplay(display === "0" ? d : display + d);
  };
  const inputDot = () => {
    if (waiting) { setDisplay("0."); setWaiting(false); return; }
    if (!display.includes(".")) setDisplay(display + ".");
  };
  const clearAll = () => { setDisplay("0"); setStored(null); setOperator(null); setWaiting(false); };
  const backspace = () => setDisplay(display.length > 1 ? display.slice(0, -1) : "0");
  const toggleSign = () => setDisplay((parseFloat(display) * -1).toString());

  const compute = (a, op, b) => {
    switch (op) {
      case "+": return a + b;
      case "−": return a - b;
      case "×": return a * b;
      case "÷": return b === 0 ? 0 : a / b;
      case "xʸ": return Math.pow(a, b);
      default: return b;
    }
  };

  const chooseOperator = (op) => {
    const cur = parseFloat(display);
    if (stored !== null && operator && !waiting) {
      const result = compute(stored, operator, cur);
      setDisplay(String(round(result)));
      setStored(round(result));
    } else {
      setStored(cur);
    }
    setOperator(op);
    setWaiting(true);
  };

  const equals = () => {
    if (operator === null || stored === null) return;
    const cur = parseFloat(display);
    const result = compute(stored, operator, cur);
    setDisplay(String(round(result)));
    setStored(null);
    setOperator(null);
    setWaiting(true);
  };

  const round = (n) => Math.round(n * 1e10) / 1e10;

  const unary = (fn) => {
    const cur = parseFloat(display);
    setDisplay(String(round(fn(cur))));
    setWaiting(true);
  };

  const deg2rad = (d) => (d * Math.PI) / 180;

  const BUTTONS = [

/* ======================================================================
   DATE: AGE CALCULATOR
   ====================================================================== */
function AgeCalc() {
  const today = new Date().toISOString().slice(0, 10);
  const [dob, setDob] = useState("1995-06-15");
  const [target, setTarget] = useState(today);

  const result = useMemo(() => {
    const start = new Date(dob + "T00:00:00");
    const end = new Date(target + "T00:00:00");
    if (isNaN(start) || isNaN(end) || end < start) return null;
    let y = end.getFullYear() - start.getFullYear();
    let m = end.getMonth() - start.getMonth();
    let d = end.getDate() - start.getDate();
    if (d < 0) { m -= 1; d += new Date(end.getFullYear(), end.getMonth(), 0).getDate(); }
    if (m < 0) { y -= 1; m += 12; }
    const totalDays = Math.round((end - start) / 86400000);
    const totalWeeks = Math.floor(totalDays / 7);
    return { y, m, d, totalDays, totalWeeks };
  }, [dob, target]);

  return (
    <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
      <Card>
        <div className="space-y-5">
          <Field label="Date of Birth"><input type="date" value={dob} onChange={(e) => setDob(e.target.value)} className={inputClass} /></Field>
          <Field label="Target Date"><input type="date" value={target} onChange={(e) => setTarget(e.target.value)} className={inputClass} /></Field>
        </div>
      </Card>

      <div className="space-y-4">
        {result ? (
          <>
            <DigitalReadout label="Exact Age" value={`${result.y}y ${result.m}m ${result.d}d`} big />
            <Card>
              <div className="grid grid-cols-2 gap-4 text-sm">
                <div><div className="text-[var(--ink-muted)]">Total Weeks</div><div className="text-lg font-semibold">{fmt(result.totalWeeks, 0)}</div></div>
                <div><div className="text-[var(--ink-muted)]">Total Days</div><div className="text-lg font-semibold">{fmt(result.totalDays, 0)}</div></div>
              </div>
            </Card>
          </>
        ) : (
          <Card><p className="text-sm text-[var(--ink-muted)]">Target date must be on or after the date of birth.</p></Card>
        )}
      </div>

      <ExplainSection title="How age is broken down">
        <p>Years, months, and days are calculated calendar-aware — borrowing days from the previous month and months from the previous year exactly as a wall calendar would.</p>
        <p>Total weeks and total days are a straight count of elapsed 24-hour periods between the two dates.</p>
      </ExplainSection>
    </div>
  );
}

/* ======================================================================
   DATE: DATE DIFFERENCE
   ====================================================================== */
function DateDiffCalc() {
  const today = new Date().toISOString().slice(0, 10);
  const [start, setStart] = useState(today);
  const [end, setEnd] = useState(new Date(Date.now() + 45 * 86400000).toISOString().slice(0, 10));

  const result = useMemo(() => {
    const s = new Date(start + "T00:00:00");
    const e = new Date(end + "T00:00:00");
    if (isNaN(s) || isNaN(e) || e < s) return null;
    const totalDays = Math.round((e - s) / 86400000);
    let business = 0;
    const cursor = new Date(s);
    for (let i = 0; i < totalDays; i++) {
      const day = cursor.getDay();
      if (day !== 0 && day !== 6) business++;
      cursor.setDate(cursor.getDate() + 1);
    }
    const weeks = Math.floor(totalDays / 7);
    const now = new Date();
    now.setHours(0, 0, 0, 0);
    const remaining = Math.round((e - now) / 86400000);
    return { totalDays, business, weeks, remainDays: totalDays % 7, remaining };
  }, [start, end]);

  return (
    <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
      <Card>
        <div className="space-y-5">
          <Field label="Start Date"><input type="date" value={start} onChange={(e) => setStart(e.target.value)} className={inputClass} /></Field>
          <Field label="End Date"><input type="date" value={end} onChange={(e) => setEnd(e.target.value)} className={inputClass} /></Field>
        </div>
      </Card>

      <div className="space-y-4">
        {result ? (
          <>
            <DigitalReadout label="Total Duration" value={`${result.weeks}w ${result.remainDays}d`} sub={`${fmt(result.totalDays, 0)} total days`} big />
            <Card>
              <div className="grid grid-cols-2 gap-4 text-sm">
                <div><div className="text-[var(--ink-muted)]">Business Days</div><div className="text-lg font-semibold">{fmt(result.business, 0)}</div></div>
                <div>
                  <div className="text-[var(--ink-muted)]">{result.remaining >= 0 ? "Time Remaining" : "Time Since End"}</div>
                  <div className="text-lg font-semibold">{fmt(Math.abs(result.remaining), 0)} days</div>
                </div>
              </div>
            </Card>
          </>
        ) : (
          <Card><p className="text-sm text-[var(--ink-muted)]">End date must be on or after the start date.</p></Card>
        )}
      </div>

      <ExplainSection title="How the duration is calculated">
        <p>Total days is a direct count between the two calendar dates. Business days exclude every Saturday and Sunday in that range.</p>
        <p>Time remaining compares the end date against today, showing a countdown for future dates or elapsed time for past ones.</p>
      </ExplainSection>
    </div>
  );
}

/* ======================================================================
   CONVERTER: UNIT CONVERTER
   ====================================================================== */
const UNIT_GROUPS = {
  length: { label: "Length", base: "m", units: { mm: 0.001, cm: 0.01, m: 1, km: 1000, in: 0.0254, ft: 0.3048, yd: 0.9144, mi: 1609.34 } },
  weight: { label: "Weight", base: "kg", units: { mg: 0.000001, g: 0.001, kg: 1, t: 1000, oz: 0.0283495, lb: 0.453592 } },
  speed: { label: "Speed", base: "m/s", units: { "m/s": 1, "km/h": 0.277778, mph: 0.44704, knot: 0.514444, "ft/s": 0.3048 } },
  temperature: { label: "Temperature", base: "C", units: { C: "C", F: "F", K: "K" } },
};

function convertTemp(v, from, to) {
  let c;
  if (from === "C") c = v;
  else if (from === "F") c = ((v - 32) * 5) / 9;
  else c = v - 273.15;
  if (to === "C") return c;
  if (to === "F") return (c * 9) / 5 + 32;
  return c + 273.15;
}

function UnitConverterCalc() {
  const [category, setCategory] = useState("length");
  const group = UNIT_GROUPS[category];
  const unitKeys = Object.keys(group.units);
  const [from, setFrom] = useState(unitKeys[0]);
  const [to, setTo] = useState(unitKeys[1]);
  const [value, setValue] = useState("1");

  useEffect(() => {
    const keys = Object.keys(UNIT_GROUPS[category].units);
    setFrom(keys[0]);
    setTo(keys[1]);
  }, [category]);

  const result = useMemo(() => {
    const v = toNum(value);
    if (category === "temperature") return convertTemp(v, from, to);
    const base = v * (group.units[from] ?? 1);
    return base / (group.units[to] ?? 1);
  }, [value, from, to, category, group]);

  return (
    <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
      <Card>
        <div className="space-y-5">
          <Field label="Category">
            <SelectInput value={category} onChange={setCategory} options={Object.entries(UNIT_GROUPS).map(([k, g]) => ({ value: k, label: g.label }))} />
          </Field>
          <Field label="Value"><NumberInput value={value} onChange={setValue} /></Field>
          <div className="grid grid-cols-[1fr,auto,1fr] items-end gap-2">
            <Field label="From">
              <SelectInput value={from} onChange={setFrom} options={unitKeys.map((u) => ({ value: u, label: u }))} />
            </Field>
            <button
              onClick={() => { setFrom(to); setTo(from); }}
              className="mb-0.5 flex h-10 w-10 items-center justify-center rounded-full border border-[var(--border)] bg-[var(--surface-2)] hover:border-[var(--brass)]"
            >
              <ArrowLeftRight className="h-4 w-4" />
            </button>
            <Field label="To">
              <SelectInput value={to} onChange={setTo} options={unitKeys.map((u) => ({ value: u, label: u }))} />
            </Field>
          </div>
        </div>
      </Card>

      <div className="space-y-4">
        <DigitalReadout label={`${value || 0} ${from} equals`} value={`${fmt(result, 4)} ${to}`} big />
        <Card>
          <p className="text-sm text-[var(--ink-muted)]">
            Conversions use fixed standard factors relative to a common base unit ({category === "temperature" ? "Celsius" : group.base}), so any pair within the {group.label.toLowerCase()} category can be combined.
          </p>
        </Card>
      </div>

      <ExplainSection title="How conversions work">
        <p>For length, weight, and speed, every unit is converted to a common base unit first, then converted from the base into the target unit — this keeps every pair of units mutually consistent.</p>
        <p>Temperature is handled separately since Celsius, Fahrenheit, and Kelvin scales don't share a simple multiplicative factor.</p>
      </ExplainSection>
    </div>
  );
}
